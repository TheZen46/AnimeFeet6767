$\begin{cases}x=R\cos(\omega t)\\y=R\sin(\omega t)\end{cases}$

$\begin{cases}x=R\cos(\theta(t))\\y=R\sin(\theta(t))\end{cases}$

$x^2+y^2=R^2\ \ \ \ \text{sempre su circonferenza}$

$\begin{cases}v_x=-R\frac{d\theta}{dt}\sin(\theta(t))\\v_y=R\frac{d\theta}{dt}\cos(\theta(t))\end{cases}$

$\frac{d\theta}{dt}\equiv\text{Velocità angolare}$

$\vec v\perp\vec r$

$\begin{cases}a_x=-R\frac{d^2\theta}{dt^2}\sin(\theta(t))-R\frac{d^2\theta}{dt^2}\cos(\theta(t))\\a_y=R\frac{d^2\theta}{dt^2}\cos(\theta(t))-R\frac{d^2\theta}{dt^2}\sin(\theta(t))\end{cases}$

$\frac{d^2\theta}{dt^2}=\alpha\ (\text{accelerazione angolare})$
