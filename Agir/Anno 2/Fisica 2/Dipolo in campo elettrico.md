$r=\text{distanza tra i poli, }d=\text{distanza di un polo dal centro}$

$\vec F=q\vec E$
	$F_+=qE$
	$F_-=(-q)E$
$\vec E$ verso destra, $\oplus$ (sopra) ha forza verso destra, mentre $\ominus$ (sotto) verso sinistra
Momento torcente

$\vec\tau=\vec r\times\vec F$

$\vec\tau_+=\vec r\times\vec F_+$
	$\vec F=\vec F_{\small{\rlap{/}\ /}}+\vec F_\perp$
$|\vec A\times\vec B|=|A||B|\sin\theta$
$\vec r\times\vec F_{\small{\rlap{/}\ /}}=0$
$|\vec r\times\vec F_\perp|=rF_\perp$
	$F_\perp=|F|\sin\theta$
$\displaystyle |\vec\tau_+|=r|F|\sin\theta=qEr\sin\theta=\frac{qEd}{2}\sin\theta$

$\vec\tau_-=\vec r\times\vec F_-$
	$\vec F=\vec F_{\small{\rlap{/}\ /}}+\vec F_\perp$
$|\vec A\times\vec B|=|A||B|\sin\theta$
$\vec r\times\vec F_{\small{\rlap{/}\ /}}=0$
$|\vec r\times\vec F_\perp|=rF_\perp$
	$F_\perp=|F|\sin\theta$
$\displaystyle |\vec\tau_-|=r|F|\sin\theta=qEr\sin\theta=\frac{qEd}{2}\sin\theta$

$\displaystyle |\tau|=\tau_++\tau_-=\cancel 2\frac{qd}{\cancel 2}E\sin\theta=qdE\sin\theta$



$\displaystyle \vec F=ds=-\frac{\tau}{d}ds$
$\displaystyle L=\int_i^\gamma\vec F\cdot d\vec s=\int_i^\gamma Fds\cos(90\degree-\theta)=\int_i^\gamma Fds\sin\theta=\int_i^\gamma qE\sin\theta ds$
